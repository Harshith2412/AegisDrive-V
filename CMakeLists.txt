cmake_minimum_required(VERSION 3.20)
project(AegisDriveV VERSION 1.0.0 LANGUAGES CXX)

# CMake validated on macOS; v1.0.0 review evidence is isolated under results/v1.0.0.

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE RelWithDebInfo)
endif()

option(AEGIS_BUILD_TESTS "Build C++ unit tests" ON)
option(AEGIS_USE_MINITEST "Use the bundled minitest fallback instead of fetching GoogleTest" ON)
option(AEGIS_MCDC "Request LLVM MC/DC instrumentation; requires supported compiler" OFF)
option(AEGIS_LLVM_COVERAGE "LLVM source-based line/region/branch coverage" OFF)
option(AEGIS_COVERAGE "Instrument for statement/branch coverage" OFF)

set(AEGIS_WARNINGS -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)

add_library(aegis_ecu_core STATIC
  ecu/src/e2e.cpp ecu/src/codec.cpp ecu/src/adas_ecu.cpp ecu/src/brake_ecu.cpp ecu/src/body_ecu.cpp)
target_include_directories(aegis_ecu_core PUBLIC ecu/include)
target_compile_options(aegis_ecu_core PRIVATE ${AEGIS_WARNINGS})
set_target_properties(aegis_ecu_core PROPERTIES POSITION_INDEPENDENT_CODE ON)

add_library(aegis_ecu SHARED ecu/src/aegis_c_api.cpp)
target_link_libraries(aegis_ecu PRIVATE aegis_ecu_core)
target_compile_options(aegis_ecu PRIVATE ${AEGIS_WARNINGS})
set_target_properties(aegis_ecu PROPERTIES CXX_VISIBILITY_PRESET hidden
  LIBRARY_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR} RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR})

if(AEGIS_MCDC)
  if(NOT AEGIS_LLVM_COVERAGE)
    message(FATAL_ERROR "MC/DC requires LLVM source coverage")
  endif()
  include(CheckCXXCompilerFlag)
  set(CMAKE_REQUIRED_LINK_OPTIONS -fprofile-instr-generate)
  check_cxx_compiler_flag("-fcoverage-mcdc -fprofile-instr-generate -fcoverage-mapping" AEGIS_HAS_MCDC)
  unset(CMAKE_REQUIRED_LINK_OPTIONS)
  if(NOT AEGIS_HAS_MCDC)
    message(FATAL_ERROR "Installed compiler does not support requested MC/DC flags")
  endif()
  target_compile_options(aegis_ecu_core PRIVATE -fcoverage-mcdc)
  target_compile_options(aegis_ecu PRIVATE -fcoverage-mcdc)
endif()
if(AEGIS_LLVM_COVERAGE AND AEGIS_COVERAGE)
  message(FATAL_ERROR "Select one coverage backend")
endif()
if(AEGIS_LLVM_COVERAGE)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    message(FATAL_ERROR "LLVM coverage requires Clang")
  endif()
  foreach(t aegis_ecu_core aegis_ecu)
    target_compile_options(${t} PRIVATE -fprofile-instr-generate -fcoverage-mapping -O0)
    target_link_options(${t} PUBLIC -fprofile-instr-generate)
  endforeach()
endif()
if(AEGIS_COVERAGE)
  target_compile_options(aegis_ecu_core PRIVATE --coverage -O0)
  target_link_options(aegis_ecu_core PUBLIC --coverage)
endif()

if(AEGIS_BUILD_TESTS)
  enable_testing()
  file(GLOB AEGIS_TEST_SOURCES CONFIGURE_DEPENDS ecu/tests/*.cpp)
  if(AEGIS_USE_MINITEST)
    add_executable(aegis_ecu_tests ${AEGIS_TEST_SOURCES} third_party/minitest/minitest_main.cpp)
    target_include_directories(aegis_ecu_tests PRIVATE third_party/minitest)
  else()
    include(FetchContent)
    FetchContent_Declare(googletest
      URL https://github.com/google/googletest/archive/refs/tags/v1.14.0.tar.gz
      DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
    set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
    set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
    FetchContent_MakeAvailable(googletest)
    add_executable(aegis_ecu_tests ${AEGIS_TEST_SOURCES})
    target_link_libraries(aegis_ecu_tests PRIVATE GTest::gtest_main)
  endif()
  target_link_libraries(aegis_ecu_tests PRIVATE aegis_ecu_core)
  target_compile_options(aegis_ecu_tests PRIVATE -Wall -Wextra)
  if(AEGIS_LLVM_COVERAGE)
    # Instrument test translation units too: inline ECU methods must have consistent
    # profile definitions across the executable and shared library.
    target_compile_options(aegis_ecu_tests PRIVATE -fprofile-instr-generate -fcoverage-mapping -O0)
    if(AEGIS_MCDC)
      target_compile_options(aegis_ecu_tests PRIVATE -fcoverage-mcdc)
    endif()
  endif()
  add_test(NAME aegis_ecu_tests COMMAND aegis_ecu_tests)
endif()
